// packages/backend/core/src/llm/LLMService.ts

import { ExponentialBackoff } from '../utils/retry';

export class LLMService {
  private readonly retryConfig = {
    maxRetries: 3,
    baseDelayMs: 1000,
    maxDelayMs: 10000,
  };

  /**
   * 带重试的生成方法
   */
  async generate(request: LLMRequest): Promise<LLMResponse> {
    const backoff = new ExponentialBackoff(this.retryConfig);
    
    return backoff.execute(async () => {
      // 现有生成逻辑
      return this.generateInternal(request);
    }, {
      shouldRetry: (error) => this.isRetryableError(error),
      onRetry: (attempt, error) => {
        this.logger.warn('LLM call failed, retrying', { attempt, error: error.message });
      },
    });
  }

  /**
   * 真正的流式生成
   */
  async *generateStream(request: LLMRequest): AsyncGenerator<string> {
    const provider = this.providers.get(request.provider ?? 'openai');
    if (!provider || !provider.generateStream) {
      // 降级到非流式
      const response = await this.generate(request);
      yield response.content;
      return;
    }

    for await (const chunk of provider.generateStream(request)) {
      yield chunk;
    }
  }

  private isRetryableError(error: Error): boolean {
    const retryableCodes = ['429', '503', '504', 'ECONNRESET', 'ETIMEDOUT'];
    return retryableCodes.some(code => error.message.includes(code));
  }
}
// packages/backend/core/src/utils/retry.ts (新文件)

export interface RetryConfig {
  maxRetries: number;
  baseDelayMs: number;
  maxDelayMs: number;
}

export class ExponentialBackoff {
  constructor(private config: RetryConfig) {}

  async execute<T>(
    fn: () => Promise<T>,
    options: {
      shouldRetry?: (error: Error) => boolean;
      onRetry?: (attempt: number, error: Error) => void;
    } = {},
  ): Promise<T> {
    let lastError: Error;
    
    for (let attempt = 0; attempt <= this.config.maxRetries; attempt++) {
      try {
        return await fn();
      } catch (error) {
        lastError = error as Error;
        
        if (attempt === this.config.maxRetries) break;
        if (options.shouldRetry && !options.shouldRetry(lastError)) break;
        
        options.onRetry?.(attempt + 1, lastError);
        
        const delay = Math.min(
          this.config.baseDelayMs * Math.pow(2, attempt),
          this.config.maxDelayMs,
        );
        await this.sleep(delay);
      }
    }
    
    throw lastError!;
  }

  private sleep(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

// packages/backend/core/src/self-healing/IntentTracker.ts

export class IntentTracker {
  private readonly intents: Map<string, TestIntent> = new Map();
  private readonly llmService?: LLMService;

  /**
   * 记录测试意图 - 完整实现
   */
  async recordIntent(
    testName: string,
    testFile: string,
    actionType: ActionType,
    selector: string,
    element: ElementHandle,
    options: IntentRecordOptions = {},
  ): Promise<TestIntent> {
    const features = await this.extractElementFeatures(element);
    
    let description = options.description;
    if (!description && options.autoGenerateDescription && this.llmService) {
      description = await this.generateIntentDescription(actionType, features);
    }

    const intent: TestIntent = {
      id: generateUUID(),
      testName,
      testFile,
      actionType,
      selector,
      elementFeatures: features,
      description,
      recordedAt: new Date(),
      confidence: 1.0,
    };

    const key = this.buildIntentKey(testName, selector);
    this.intents.set(key, intent);
    
    return intent;
  }

  /**
   * 根据意图重新定位元素
   */
  async relocateByIntent(
    intent: TestIntent,
    context: FixContext,
  ): Promise<RelocateResult | null> {
    const strategies = [
      () => this.relocateById(intent, context),
      () => this.relocateByText(intent, context),
      () => this.relocateByAttributes(intent, context),
      () => this.relocateByPosition(intent, context),
    ];

    for (const strategy of strategies) {
      const result = await strategy();
      if (result && result.confidence >= 0.7) {
        return result;
      }
    }

    return null;
  }

  private async extractElementFeatures(element: ElementHandle): Promise<ElementFeatures> {
    // 提取元素特征用于后续重定位
    return {
      tagName: await element.evaluate(el => el.tagName.toLowerCase()),
      id: await element.evaluate(el => el.id),
      className: await element.evaluate(el => el.className),
      textContent: await element.evaluate(el => el.textContent?.trim().slice(0, 100)),
      attributes: await element.evaluate(el => {
        const attrs: Record<string, string> = {};
        for (const attr of el.attributes) {
          attrs[attr.name] = attr.value;
        }
        return attrs;
      }),
      boundingBox: await element.boundingBox(),
    };
  }
}
2.4 Diff 算法优化 (优先级: 中)
问题描述
当前使用简化的 diff 算法，对于复杂代码变更可能产生不准确的结果。

修复方案
// packages/backend/core/src/diff/DiffGenerator.ts

import { diffLines, Change } from 'diff'; // 使用成熟的 diff 库

export class DiffGenerator {
  /**
   * 使用 Myers diff 算法计算差异
   */
  private computeDiff(
    oldLines: string[],
    newLines: string[],
  ): Array<{ type: 'add' | 'delete' | 'equal'; oldIndex: number; newIndex: number }> {
    const oldText = oldLines.join('\n');
    const newText = newLines.join('\n');
    
    const changes = diffLines(oldText, newText);
    const result: Array<{ type: 'add' | 'delete' | 'equal'; oldIndex: number; newIndex: number }> = [];
    
    let oldIndex = 0;
    let newIndex = 0;

    for (const change of changes) {
      const lines = change.value.split('\n').filter(l => l !== '' || change.value === '\n');
      
      for (let i = 0; i < lines.length; i++) {
        if (change.added) {
          result.push({ type: 'add', oldIndex: -1, newIndex: newIndex++ });
        } else if (change.removed) {
          result.push({ type: 'delete', oldIndex: oldIndex++, newIndex: -1 });
        } else {
          result.push({ type: 'equal', oldIndex: oldIndex++, newIndex: newIndex++ });
        }
      }
    }

    return result;
  }
}

import { describe, it, expect, beforeEach, vi } from 'vitest';

describe('SelfHealingEngine', () => {
